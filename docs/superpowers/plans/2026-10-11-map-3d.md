## Global Constraints

- No Minecraft types in `dev.moui.galaxycraft.map` (so it is unit tested).
- Never edit `ship/*` or `ShipClient`; the toad ship session owns them.
- Universe positions are `UPos`; the map works in **units relative to Mario's universe position** (small doubles), never absolute doubles.
- Blocks per unit: `1 / GravityFrame.SCALE` (80 by default); tests pass `80` explicitly.
- `courses.json` sits in the world's planets folder next to `ships.json`; a damaged file is dropped, never crashes.
- Nothing in the map starts Dolphin; never run gxplay/gxvoxel while another session uses it.
- Tests: `cd fabric && ./test.sh --tests '<Class>'` (JDK 25 is picked from `~/.local/opt/jdk-25*`). Compile client: `cd fabric && JAVA_HOME=$(ls -d ~/.local/opt/jdk-25*|head -1) ./gradlew compileClientJava --console=plain -q`.
- Commit messages end with `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.

## Review Focus

- A ship sent to where it already is (zero-length course): arrives within a tick, no NaN, then orbits.
- A course replaced in flight starts from the ship's position *now*, not from its old origin.
- Two ships sent to the same planet get different orbit phases.
- Missing, empty or corrupt `courses.json`: empty store, no exception.
- Map opened with no universe, no other system in reach, or Mario far from the origin (huge cells): no crash, no precision jitter.
- Projection: a point behind the camera is not drawn; a resized window recenters.

```java
import dev.moui.galaxycraft.gravity.GravityFrame;
import dev.moui.galaxycraft.universe.UPos;
import dev.moui.galaxycraft.universe.Universe;

/**
 * Tier and star of a system, for the map and for anyone who needs them (Toad NPC dialogue). Today
 * a stand-in: the tier from the distance to home (the progression spec's rings), the star always
 * yellow. When SystemIndex carries real values only this class changes.
 */
public record SystemInfo(int tier, StarType star) {
    /** Ring edges, blocks from home: tier 0 up to the first (home and its neighbors), 5 past the last. First draft. */
    static final double[] EDGES = {12_000, 20_000, 40_000, 70_000, 110_000};

    public static SystemInfo of(Universe.Star star) {
        return of(star, 1 / GravityFrame.SCALE);
    }

    public static SystemInfo of(Universe.Star star, double unitsPerBlock) {
        if (star.home()) return new SystemInfo(0, StarType.YELLOW);
        double blocks = star.center().minus(UPos.ZERO).length() / unitsPerBlock;
        return new SystemInfo(tierAt(blocks), StarType.YELLOW);
    }

    public static int tierAt(double blocksFromHome) {
        int t = 0;
        for (double e : EDGES) if (blocksFromHome > e) t++;
        return t;
    }
}
```

```java
package dev.moui.galaxycraft.map;

import java.util.List;
import java.util.Optional;
import org.joml.Vector3d;

/**
 * The map's orbit camera: looks at a focus from a distance, turned by yaw and pitch. Projects
 * world points (units, relative to the player) to the screen and picks the nearest point to the
 * mouse. No Minecraft types.
 */
public final class MapView {
    public record Screen(double x, double y, double depth) {}

    private static final double MAX_PITCH = 1.5, FOV = Math.toRadians(60), MIN_DIST = 1e-3;

    public int width = 320, height = 240;
    private final Vector3d focus = new Vector3d();
    private double yaw = 0.6, pitch = 0.5, distance = 1;

    public Vector3d focus() {
        return new Vector3d(focus);
    }

    public double distance() {
        return distance;
    }

    public void focusOn(Vector3d f, double dist) {
        focus.set(f);
        distance = Math.max(MIN_DIST, dist);
    }

    /** Drag: radians. */
    public void orbit(double dYaw, double dPitch) {
        yaw += dYaw;
        pitch = Math.max(-MAX_PITCH, Math.min(MAX_PITCH, pitch + dPitch));
    }

    public void zoom(double factor) {
        distance = Math.max(MIN_DIST, Math.min(1e15, distance * factor));
    }

    public Vector3d eye() {
        double cp = Math.cos(pitch);
        return new Vector3d(focus).add(distance * cp * Math.sin(yaw), distance * Math.sin(pitch), distance * cp * Math.cos(yaw));
    }

    public Optional<Screen> project(Vector3d p) {
        Vector3d eye = eye();
        Vector3d f = new Vector3d(focus).sub(eye).normalize();
        Vector3d r = new Vector3d(f).cross(0, 1, 0).normalize();
        Vector3d u = new Vector3d(r).cross(f);
        Vector3d d = new Vector3d(p).sub(eye);
        double z = d.dot(f);
        if (z <= distance * 1e-6) return Optional.empty();
        double scale = (height / 2.0) / Math.tan(FOV / 2);
        return Optional.of(new Screen(width / 2.0 + d.dot(r) / z * scale, height / 2.0 - d.dot(u) / z * scale, z));
    }

    /** The index of the point nearest the mouse within radiusPx on screen (ties: the nearer to the camera), or -1. */
    public int pick(List<Vector3d> points, double mx, double my, double radiusPx) {
        int best = -1;
        double bestD = radiusPx * radiusPx, bestZ = Double.MAX_VALUE;
        for (int i = 0; i < points.size(); i++) {
            Optional<Screen> s = project(points.get(i));
            if (s.isEmpty()) continue;
            double dx = s.get().x() - mx, dy = s.get().y() - my, d = dx * dx + dy * dy;
            if (d < bestD || (d == bestD && s.get().depth() < bestZ)) {
                best = i;
                bestD = d;
                bestZ = s.get().depth();
            }
        }
        return best;
    }
}
```

```java
package dev.moui.galaxycraft.map;

import dev.moui.galaxycraft.universe.UPos;
import org.joml.Vector3d;

/** A straight trip: from, to (universe), the tick it left and its speed, units a tick (> 0). Position is a function of the tick. */
public record Course(UPos from, UPos to, long startTick, double speed) {
    public double length() {
        return to.minus(from).length();
    }

    public long durationTicks() {
        return (long) Math.ceil(length() / speed);
    }

    public boolean arrived(long t) {
        return t - startTick >= durationTicks();
    }

    public UPos position(long t) {
        long d = durationTicks();
        if (d == 0 || t - startTick >= d) return to;
        if (t <= startTick) return from;
        Vector3d step = to.minus(from).mul((double) (t - startTick) / d);
        return from.plus(step);
    }
}
```

```java
package dev.moui.galaxycraft.map;

import dev.moui.galaxycraft.universe.UPos;
import org.joml.Vector3d;

/**
 * A circle around a planet (in the universe's x-z plane): radius units, phase radians at
 * epochTick, rate radians a tick.
 */
public record Orbit(UPos center, double radius, double phase, double rate, long epochTick) {
    public UPos position(long t) {
        double a = phase + rate * (t - epochTick);
        return center.plus(new Vector3d(radius * Math.cos(a), 0, radius * Math.sin(a)));
    }
}
```

```java
package dev.moui.galaxycraft.map;

import dev.moui.galaxycraft.universe.UPos;

/** A ship's whole trip: the course there, then the orbit it ends in. Nothing is simulated: ask for any tick. */
public record Trip(Course course, Orbit orbit) {
    public boolean arrived(long t) {
        return course.arrived(t);
    }

    public UPos positionAt(long t) {
        return arrived(t) ? orbit.position(t) : course.position(t);
    }
}
```

```java
            JsonObject o = JsonParser.parseString(Files.readString(file)).getAsJsonObject();
            Map<String, Trip> read = new LinkedHashMap<>();
            for (Map.Entry<String, JsonElement> e : o.entrySet()) {
                JsonObject t = e.getValue().getAsJsonObject();
                JsonObject c = t.getAsJsonObject("course"), r = t.getAsJsonObject("orbit");
                Course course = new Course(pos(c.getAsJsonArray("from")), pos(c.getAsJsonArray("to")), c.get("start").getAsLong(),
                        c.get("speed").getAsDouble());
                Orbit orbit = new Orbit(pos(r.getAsJsonArray("center")), r.get("radius").getAsDouble(), r.get("phase").getAsDouble(),
                        r.get("rate").getAsDouble(), r.get("epoch").getAsLong());
                read.put(e.getKey(), new Trip(course, orbit));
            }
            s.trips.putAll(read);
        } catch (IOException | RuntimeException e) {
            s.trips.clear();
        }
        return s;
    }

    public Optional<Trip> get(String id) {
        return Optional.ofNullable(trips.get(id));
    }

    public void put(String id, Trip t) {
        trips.put(id, t);
    }

    public Optional<Trip> remove(String id) {
        return Optional.ofNullable(trips.remove(id));
    }

    public Map<String, Trip> all() {
        return Map.copyOf(trips);
    }

    /** How many trips end orbiting this center (to give each ship its own phase). */
    public int orbitingAround(UPos center) {
        int n = 0;
        for (Trip t : trips.values()) if (t.orbit().center().equals(center)) n++;
        return n;
    }

    public void save() throws IOException {
        JsonObject o = new JsonObject();
        for (Map.Entry<String, Trip> e : trips.entrySet()) {
            Trip t = e.getValue();
            JsonObject c = new JsonObject(), r = new JsonObject(), j = new JsonObject();
            c.add("from", arr(t.course().from()));
            c.add("to", arr(t.course().to()));
            c.addProperty("start", t.course().startTick());
            c.addProperty("speed", t.course().speed());
            r.add("center", arr(t.orbit().center()));
            r.addProperty("radius", t.orbit().radius());
            r.addProperty("phase", t.orbit().phase());
            r.addProperty("rate", t.orbit().rate());
            r.addProperty("epoch", t.orbit().epochTick());
            j.add("course", c);
            j.add("orbit", r);
            o.add(e.getKey(), j);
        }
        Files.createDirectories(file.toAbsolutePath().getParent());
        Files.writeString(file, o.toString());
    }

    private static JsonArray arr(UPos p) {
        JsonArray a = new JsonArray();
        a.add(p.cx());
        a.add(p.cy());
        a.add(p.cz());
        a.add(p.x());
        a.add(p.y());
        a.add(p.z());
        return a;
    }

    private static UPos pos(JsonArray a) {
        if (a.size() != 6) throw new IllegalArgumentException("position needs 6 numbers");
        return new UPos(a.get(0).getAsLong(), a.get(1).getAsLong(), a.get(2).getAsLong(), a.get(3).getAsDouble(),
                a.get(4).getAsDouble(), a.get(5).getAsDouble());
    }
```

```java
package dev.moui.galaxycraft.map;

import dev.moui.galaxycraft.universe.UPos;
import org.joml.Vector3d;

/** Sending a ship to a planet and stopping it: pure, on a CourseStore. */
public final class RemoteControl {
    /** Orbit radius over the surface, blocks: 60% of the radius plus 24. First draft, tuned by playing. */
    static double orbitBlocks(double planetRadiusBlocks) {
        return planetRadiusBlocks * 1.6 + 24;
    }

    /** Radians a tick: a lap takes about 60 seconds (1200 ticks). */
    static final double RATE = 2 * Math.PI / 1200;

    private RemoteControl() {}

    /** Where the ship is at now: on its trip if it has one, else where it was recorded. */
    public static UPos positionNow(CourseStore store, String shipId, UPos recorded, long now) {
        return store.get(shipId).map(t -> t.positionAt(now)).orElse(recorded);
    }

    /**
     * A trip from where the ship is now to an orbit around the planet, kept in the store (a trip it
     * had is replaced). The orbit's phase starts on the side the ship comes from, turned by 45
     * degrees for each ship already bound for that planet so none shares a spot.
     */
    public static Trip send(CourseStore store, String shipId, UPos shipNow, UPos planetCenter, double planetRadiusBlocks,
            double unitsPerBlock, double unitsPerTick, long now) {
        UPos from = positionNow(store, shipId, shipNow, now);
        store.remove(shipId);
        double radius = orbitBlocks(planetRadiusBlocks) * unitsPerBlock;
        Vector3d side = from.minus(planetCenter);
        double base = side.x == 0 && side.z == 0 ? 0 : Math.atan2(side.z, side.x);
        double phase = base + store.orbitingAround(planetCenter) * Math.PI / 4;
        UPos to = planetCenter.plus(new Vector3d(radius * Math.cos(phase), 0, radius * Math.sin(phase)));
        Course course = new Course(from, to, now, unitsPerTick);
        Orbit orbit = new Orbit(planetCenter, radius, phase, RATE, now + course.durationTicks());
        Trip trip = new Trip(course, orbit);
        store.put(shipId, trip);
        return trip;
    }

    /** Stops the ship where it is: the trip is forgotten and that position returned (the ship side records it). */
    public static UPos cancel(CourseStore store, String shipId, UPos recorded, long now) {
        UPos at = positionNow(store, shipId, recorded, now);
        store.remove(shipId);
        return at;
    }
}
```

```java
        items.clear();

        Universe u = UniverseClient.universe();
        if (u == null) return;
        UPos m = mario();
        for (Universe.Star s : u.around(m, UniverseClient.STAR_SECTORS)) {
            SystemInfo info = SystemInfo.of(s);

        else planets = u.system(star, LegacyBiome.land()).entries();
        for (GalaxyCatalog.Entry e : planets) {

            items.add(new Item("Planet " + e.index() + "  r" + e.radius(), at.minus(m), e.radius() * UNITS, 0xFF8FD16B, star, at, e.radius()));

        g.fill(0, 0, width, height, 0xFF05060C);
        view.width = width;
        view.height = height;
        long now = MapClient.now();
        for (int i = 0; i < items.size(); i++) {
            Item it = items.get(i);
            var p = view.project(it.pos());
            if (p.isEmpty()) continue;
            int r = (int) Math.max(2, Math.min(24, it.size() / p.get().depth() * (height / 2.0) / Math.tan(Math.toRadians(30))));
            disc(g, (int) p.get().x(), (int) p.get().y(), r, it.color());
            if (i == selected) ring(g, (int) p.get().x(), (int) p.get().y(), r + 4, WHITE);
            if (r >= 3 || i == selected) g.text(font, it.label(), (int) p.get().x() + r + 3, (int) p.get().y() - 4, i == selected ? WHITE : GRAY);
        }
        if (inside != null) shipMarkers(g, now);
        g.fill(width - 175, 0, width, height, PANEL);
        g.centeredText(font, inside == null ? "Galaxy Map" : "System " + (inside.home() ? "Home" : SystemIndex.name(inside.sector())), width / 2, 12, WHITE);
        if (selected >= 0) info(g, items.get(selected));
        else g.text(font, items.isEmpty() ? "No system in reach" : "Click a system", width - 168, 14, GRAY);
        g.text(font, "Drag: turn   Wheel: zoom   Backspace: up", 8, height - 14, GRAY);
        super.extractRenderState(g, mouseX, mouseY, a);
    }

    private void info(GuiGraphicsExtractor g, Item it) {
        int x = width - 168, y = 14;
        g.text(font, it.label(), x, y, WHITE);
        if (it.star() != null && inside == null) {
            SystemInfo info = SystemInfo.of(it.star());
            g.text(font, "Tier " + info.tier() + ", " + info.star().name().toLowerCase().replace('_', ' ') + " star", x, y + 12, GRAY);
            g.text(font, it.star().planets() + " planets", x, y + 24, GRAY);
            g.text(font, String.format("%,d blocks", Math.round(it.pos().length() / UNITS)), x, y + 36, GRAY);
        } else {
            g.text(font, "Radius " + (int) it.radiusBlocks() + " blocks", x, y + 12, GRAY);
        }
    }

    private void shipMarkers(GuiGraphicsExtractor g, long now) {
        UPos m = mario();
        for (MapShips.Ship s : MapShips.provider().ships()) {
            Trip t = MapClient.courses().get(s.id()).orElse(null);
            UPos at = t == null ? s.position() : t.positionAt(now);
            var p = view.project(at.minus(m));
            if (p.isEmpty()) continue;
            ring(g, (int) p.get().x(), (int) p.get().y(), 4, RING);
            g.text(font, s.name() + (t == null ? "" : t.arrived(now) ? " (orbiting)" : " (en route)"), (int) p.get().x() + 7, (int) p.get().y() - 4, RING);
            if (t != null && !t.arrived(now)) {
                var d = view.project(t.course().to().minus(m));
                d.ifPresent(q -> line(g, (int) p.get().x(), (int) p.get().y(), (int) q.x(), (int) q.y(), RING));
            }
        }
    }

    private static void disc(GuiGraphicsExtractor g, int cx, int cy, int r, int color) {
        for (int dy = -r; dy <= r; dy++) {
            int w = (int) Math.sqrt((double) r * r - (double) dy * dy);
            g.fill(cx - w, cy + dy, cx + w + 1, cy + dy + 1, color);
        }
    }

    private static void ring(GuiGraphicsExtractor g, int cx, int cy, int r, int color) {
        for (int a = 0; a < 48; a++) {
            int x = cx + (int) Math.round(r * Math.cos(a * Math.PI / 24)), y = cy + (int) Math.round(r * Math.sin(a * Math.PI / 24));
            g.fill(x, y, x + 1, y + 1, color);
        }
    }

    private static void line(GuiGraphicsExtractor g, int x0, int y0, int x1, int y1, int color) {
        int n = Math.max(Math.abs(x1 - x0), Math.abs(y1 - y0));
        if (n == 0 || n > 4000) return;
        for (int i = 0; i <= n; i += 2) {
            int x = x0 + (x1 - x0) * i / n, y = y0 + (y1 - y0) * i / n;
            g.fill(x, y, x + 1, y + 1, color);
        }
    }
}
```
